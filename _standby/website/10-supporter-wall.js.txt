  // ── Supporter Reef Wall names ("Backed By People Who Care") ──────────
  // Live from the game server, the only thing allowed to read Firestore.
  // /api/supporters/wall returns ONLY approved + public supporters, already
  // sorted by lifetime spend (biggest first), each with amountCents (the
  // donor's LIFETIME total). Every name is sized CONTINUOUSLY from that total
  // so a bigger donation is always a bigger name, down to a single dollar,
  // and hovering a name shows exactly how much they've given. Empty response →
  // the illustration shows alone (no approved supporters yet).
  function escWallText(s) {
    return String(s == null ? '' : s).replace(/[&<>"']/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c];
    });
  }
  // Continuous size from the lifetime total. d/(d+K) climbs fast at the low end
  // (so $1 vs $5 is clearly visible) and saturates high (so $500 vs $1000 differ
  // only a little), it NEVER stops rising, so one extra dollar is always a
  // slightly bigger name. Fractional vw ⇒ each amount is a genuinely different
  // width. clamp keeps it readable on phones and sane on ultra-wide screens.
  function nameFontSize(cents) {
    const d = Math.max(0, (Number(cents) || 0) / 100);   // dollars
    const MIN_VW = 1.0, MAX_VW = 4.6, K = 45;
    const vw = MIN_VW + (MAX_VW - MIN_VW) * (d / (d + K));
    const capPx = (vw * 20).toFixed(1);                  // freeze growth past ~2000px-wide
    return 'clamp(9px, ' + vw.toFixed(3) + 'vw, ' + capPx + 'px)';
  }
  function moneyText(cents) {
    const n = (Number(cents) || 0) / 100;
    return '$' + (n % 1 === 0
      ? n.toLocaleString('en-US')
      : n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }));
  }
  function renderSupporterWall(list) {
    const box = document.getElementById('supporter-names');
    if (!box) return;
    if (!Array.isArray(list) || !list.length) { box.innerHTML = ''; return; }
    box.innerHTML = list.map(function (s) {
      const cents = s && s.amountCents;
      const name = escWallText(s && s.displayName);
      const amt = moneyText(cents);
      // data-tip (not title): a native title tooltip has a ~1s hover delay and
      // never appears on touch at all, data-tip feeds the instant CSS popup
      // (see .scn-name::after) instead. tabindex makes it keyboard-focusable
      // so the popup is also reachable without a mouse.
      return '<span class="scn-name" tabindex="0" style="font-size:' + nameFontSize(cents) + '"'
        + ' data-tip="Donated ' + amt + '"'
        + ' aria-label="' + name + ', donated ' + amt + '">' + name + '</span>';
    }).join('');
  }
  async function refreshSupporterWall() {
    try {
      const resp = await fetchWithTimeout(`${RENDER_BASE}/api/supporters/wall`, 30000);
      if (!resp.ok) return;
      const data = await resp.json();
      const list = data && data.supporters ? data.supporters : [];
      renderSupporterWall(list);
      // Live donation total = the server's sum of exactly these wall rows
      // (falls back to summing them here if an older server omits the field).
      // Drives the donation bar + the "$ Donated" impact stat, and is cached so
      // a returning visitor sees the real amount before the next fetch returns.
      let cents = Number(data && data.totalRaisedCents);
      if (!Number.isFinite(cents)) {
        cents = list.reduce((sum, s) => sum + (Number(s && s.amountCents) || 0), 0);
      }
      const dollars = Math.max(0, cents / 100);
      renderDonationGoal(dollars);
      const saved = readCachedStats();
      saved.donated = dollars;
      writeCachedStats(saved);
    } catch (_) { /* server sleeping, leave the scene as-is */ }
  }
  pollWhileVisible(refreshSupporterWall, LIVE_POLL_MS);

  // Touchscreens have no :hover state, so the CSS popup (.scn-name::after)
  // would never show on tap alone, this toggles a class that the CSS also
  // triggers on. Delegated on the container (names are replaced wholesale
  // every refresh, so per-name listeners would go stale).
  document.getElementById('supporter-names').addEventListener('click', function (e) {
    const el = e.target.closest('.scn-name');
    const wasOpen = el && el.classList.contains('scn-tip-open');
    this.querySelectorAll('.scn-tip-open').forEach(function (o) { o.classList.remove('scn-tip-open'); });
    if (el && !wasOpen) el.classList.add('scn-tip-open');
  });
  document.addEventListener('click', function (e) {
    const wrap = document.getElementById('supporter-names');
    if (wrap && !wrap.contains(e.target)) {
      wrap.querySelectorAll('.scn-tip-open').forEach(function (o) { o.classList.remove('scn-tip-open'); });
    }
  });
